<p class="help" id="ehelp" hidden>Endlosschlacht: Du gegen immer größere Wellen von Panzern in einem Labyrinth ohne Ende. Steuerung wie im Duell. Jeder besiegte Gegner gibt dir einen Punkt, egal wodurch er stirbt. Fähigkeiten (Tasten 1 bis 7 oder die Knöpfe unter dem Feld, die Kosten stehen auf den Knöpfen und sind auf Leicht und Mittel niedriger): Salve, Hilfsbots (ab Wave 2), Artillerie (ab Wave 3) und Elite-Bots (ab Wave 4): drei schnellere Bots, die besser ausweichen, deren Kugeln zweimal abprallen und die selbst Salven feuern. Geschütz (ab Wave 2): steht fest und schießt doppelt so schnell wie ein Panzer. Armee (ab Wave 5): 12 Bots, 2 bei dir und 10 gestaffelt vom Rand. Der mittlere Knopf (Taste E) sammelt alle deine Bots hinter dir, ein zweiter Druck schickt sie wieder los. Joker (ab Wave 3, nur einer auf Vorrat): Bei deinem Tod stirbst du nicht, sondern bist 3 Sekunden unverwundbar. Die Artillerie verfolgt 4 bis 5 Gegner, die Salve prallt auf Leicht dreimal und auf Mittel zweimal ab. Hilfsbots werden mit den Waves stärker. Auf Leicht und Mittel nutzen auch die normalen Hilfsbots Abprallwinkel, auf Leicht weichen sie zusätzlich aus. Die ersten Wellen sind langsam und ungenau, ab Wave 5 werden Gegner schneller und nutzen Abprallwinkel. Gegnertypen: Sniper (ab Wave 3, trifft auf große Entfernung und bleibt auf Abstand), Nahkämpfer (ab Wave 4, schnell, weicht gut aus, schießt schnell) und Schwere Panzer (ab Wave 6, brauchen zwei Treffer). Auf Leicht siehst du alle Gegner auf der Minikarte. Ohne Sammelbefehl teilen sich deine Hilfsbots in Gruppen zu 3 bis 5 auf und ziehen einzeln umher. Je nach Stufe feuern Gegner Salven auf dichte Truppenansammlungen. Stirbst du, ist das Spiel vorbei und du beginnst wieder bei Wave 1. Deine Bots übernimmst du nicht. Die Schwierigkeit bestimmt, wie schnell die Gegner nach Sichtkontakt reagieren und wie viele Sackgassen das Labyrinth hat (Leicht und Mittel wenige, Schwer mehr).</p>
<p class="help" id="xhelp" hidden>Basis verteidigen: Verteidige den Turm in der Mitte deiner runden Basis gegen Wellen von Angreifern. Mit dem Werkzeug Mauer erweiterst du die Mauer nach außen. Jedes Quadrat der Basis nimmt genau eine Einheit auf, in der Mitte des Quadrats. Sind alle belegt, kannst du nur noch verbessern. Von innen schießt nur die Artillerie über die Mauer. Geschütze und Tesla-Türme auf Mauerstücken erreichen Gegner innerhalb und außerhalb der Basis und gehen erst kaputt, wenn ihr Mauerstück fällt. Von jeder Waffe gibt es nur eine begrenzte Anzahl: wenig Artillerie, mehr Geschütze und Tesla-Türme, viele Soldaten. Der Tesla-Turm erfasst bis zu 3 Gegner gleichzeitig mit blauen Strahlen, wer insgesamt 2 Sekunden getroffen wird, fällt (die Kreise zeigen Reichweiten). Soldaten bewegen sich in der Basis, beziehen Posten Richtung Feind, weichen Kugeln aus und greifen ein, sobald Feinde in die Basis kommen. Auf Leicht ist der Start besonders großzügig: viel Schrott und niedrige Preise. Artillerie beschießt Feinde außerhalb der Mauer. Schrott bekommst du für Abschüsse und geschaffte Wellen. function nextCell(t,p){
  const cell=o=>[Math.min(R-1,Math.max(0,Math.floor(o.y/S))),Math.min(C-1,Math.max(0,Math.floor(o.x/S)))];
  const [sr,sc]=cell(t),[gr,gc]=cell(p);
  if(sr===gr&&sc===gc)return null;
  const start=sr*C+sc,prev=new Map([[start,-1]]),q=[[sr,sc]];
  while(q.length){
    const [r,c]=q.shift();
    if(r===gr&&c===gc)break;
    const n=[];
    if(!hw[r][c])n.push([r-1,c]);
    if(!hw[r+1][c])n.push([r+1,c]);
    if(!vw[r][c])n.push([r,c-1]);
    if(!vw[r][c+1])n.push([r,c+1]);
    for(const [a,b] of n)if(!prev.has(a*C+b)){prev.set(a*C+b,r*C+c);q.push([a,b])}
  }
  let cur=gr*C+gc;
  if(!prev.has(cur))return null;
  while(prev.get(cur)!==start)cur=prev.get(cur);
  return{x:(cur%C+.5)*S,y:(Math.floor(cur/C)+.5)*S};
}

}

function enemySetup(s){
  const all=[];
  for(let c=C/2;c<C;c++)for(let r=0;r<R;r++)all.push([c,r]);
  const used=new Set(),out=[];
  const add=(list,n,order)=>{
    for(const [c,r] of shuf(list)){

      out.push({c,r,order:typeof order==='function'?order():order});

  };

  return out;
}

function safeShot(t){
  let vx=Math.cos(t.a)*4.2,vy=Math.sin(t.a)*4.2,n=0;
  const b=muzzle(t,t.a);
  for(let s=0;s<130;s++){
    let w=false;
    b.x+=vx;if(wallAt(b.x,b.y)){b.x-=vx;vx=-vx;w=true}
    b.y+=vy;if(wallAt(b.x,b.y)){b.y-=vy;vy=-vy;w=true}
    if(w&&++n>1)return true;
    if(n>0&&Math.hypot(t.x-b.x,t.y-b.y)<15)return false;
  }
  return true;
}

function followAI(t){
  const prev=prevOf(t);
  if(!prev)return{};
  const dist=Math.hypot(prev.x-t.x,prev.y-t.y);
  t.sp=dist>150?1.9:1.5;
  if(dist<=34){
    const f=nearest(t);
    if(f&&Math.hypot(f.x-t.x,f.y-t.y)<420&&clear(t,f)){
      const e=norm(Math.atan2(f.y-t.y,f.x-t.x)-t.a);
      return{l:e<-.06,r:e>.06,s:Math.abs(e)<.1};
    }
    const e=norm(prev.a-t.a);
    return{l:e<-.06,r:e>.06};
  }
  const tg=(dist<160&&clear(t,prev))?prev:(path(t,prev)||prev);
  const d=norm(Math.atan2(tg.y-t.y,tg.x-t.x)-t.a);
  return{l:d<-.06,r:d>.06,f:Math.abs(d)<.9};
}

// Salve des Duell-Gegners: alle 30 Sekunden 3 bis 4 Kugeln im Fächer. Wann und wohin entscheidet er selbst.
function safeAng(t,ang){
  let vx=Math.cos(ang)*4.2,vy=Math.sin(ang)*4.2,n=0;
  const b=muzzle(t,ang);
  for(let s=0;s<130;s++){
    let w=false;
    b.x+=vx;if(wallAt(b.x,b.y)){b.x-=vx;vx=-vx;w=true}
    b.y+=vy;if(wallAt(b.x,b.y)){b.y-=vy;vy=-vy;w=true}
    if(w&&++n>1)return true;
    if(n>0&&Math.hypot(t.x-b.x,t.y-b.y)<15)return false;
  }
  return true;
}

function cellOf(o){return[Math.min(R-1,Math.max(0,Math.floor(o.y/S))),Math.min(C-1,Math.max(0,Math.floor(o.x/S)))]}

function flood(r0,c0){
  const seen=new Uint8Array(R*C),st=[r0*C+c0];
  seen[r0*C+c0]=1;
  while(st.length){
    const v=st.pop(),r=Math.floor(v/C),c=v%C;
    if(r>0&&!hw[r][c]&&!seen[v-C]){seen[v-C]=1;st.push(v-C)}
    if(r<R-1&&!hw[r+1][c]&&!seen[v+C]){seen[v+C]=1;st.push(v+C)}
    if(c>0&&!vw[r][c]&&!seen[v-1]){seen[v-1]=1;st.push(v-1)}
    if(c<C-1&&!vw[r][c+1]&&!seen[v+1]){seen[v+1]=1;st.push(v+1)}
  }
  return seen;
}

function extend(dir,k){

  const dc=dir==='L'?k:dir==='R'?(gr?0:-k):0,dr=dir==='U'?k:dir==='D'?(gr?0:-k):0;
  const inW=(r,c)=>r>=0&&r<R&&c>=0&&c<C;
  const isNew=(r,c)=>{const a=r-dr,b=c-dc;return !(a>=0&&a<oR&&b>=0&&b<oC)};
  const H=Array.from({length:R+1},()=>Array(C).fill(true)),V=Array.from({length:R},()=>Array(C+1).fill(true));
  const bnd=[];
  for(let r=0;r<=R;r++)for(let c=0;c<C;c++){
    const a=inW(r-1,c)?isNew(r-1,c):null,b=inW(r,c)?isNew(r,c):null;

    else if(a===false||b===false)bnd.push([H,r,c]);
  }
  for(let r=0;r<R;r++)for(let c=0;c<=C;c++){
    const a=inW(r,c-1)?isNew(r,c-1):null,b=inW(r,c)?isNew(r,c):null;
    if(a!==true&&b!==true){const or=r-dr,oc=c-dc;V[r][c]=(or>=0&&or<oR&&oc>=0&&oc<=oC)?oV[or][oc]:true}
    else if(a===false||b===false)bnd.push([V,r,c]);
  }
  const cells=[];
  for(let r=0;r<R;r++)for(let c=0;c<C;c++)if(isNew(r,c))cells.push([r,c]);
  const seen=Array.from({length:R},(_,r)=>Array.from({length:C},(_,c)=>!isNew(r,c)));
  const rm=(r,c,r2,c2)=>{if(r2<r)H[r][c]=false;else if(r2>r)H[r2][c]=false;else if(c2<c)V[r][c]=false;else V[r][c2]=false};

    if(!n.length){st.pop();continue}
    const [a,b]=n[rnd(n.length)];
    rm(r,c,a,b);seen[a][b]=true;st.push([a,b]);
  }
  for(let i=0;i<cells.length/5;i++){
    const [r,c]=cells[rnd(cells.length)],[a,b]=[[r-1,c],[r+1,c],[r,c-1],[r,c+1]][rnd(4)];
    if(inW(a,b)&&isNew(a,b))rm(r,c,a,b);
  }

  const [pr,pc]=cellOf(tanks[me]),reach=flood(pr,pc),edge=[];
  for(let r=1;r<R-1;r++)for(let c=1;c<C-1;c++)if((r===1||r===R-2||c===1||c===C-2)&&reach[r*C+c])edge.push([r,c]);
  tanks.forEach(t=>{

// Endlos-Gegner: Zielfehler und Tempo hängen von der Wave ab, Abprall-Schüsse erst ab Wave 5
// Sackgassen entfernen: Zellen mit drei Wänden bekommen mit Wahrscheinlichkeit p einen weiteren Durchgang
function braid(H,V,p,only){
  if(p<=0)return;
  for(let r=0;r<R;r++)for(let c=0;c<C;c++){
    if(only&&!only(r,c))continue;
    const w=[H[r][c],H[r+1][c],V[r][c],V[r][c+1]];
    if(w.filter(Boolean).length!==3||Math.random()>p)continue;
    const o=[];
    if(w[0]&&r>0)o.push(0);
    if(w[1]&&r<R-1)o.push(1);
    if(w[2]&&c>0)o.push(2);
    if(w[3]&&c<C-1)o.push(3);
    if(!o.length)continue;
    const d=o[rnd(o.length)];
    if(d===0)H[r][c]=false;else if(d===1)H[r+1][c]=false;else if(d===2)V[r][c]=false;else V[r][c+1]=false;
  }
}

  }
  if(p&&Math.hypot(p.x-t.x,p.y-t.y)<520&&clear(t,p)){
    if(!(now<t.on)){t.on=now+700;t.off=(Math.random()-.5)*(t.err||.3)}
    const d=norm(Math.atan2(p.y-t.y,p.x-t.x)+(t.off||0)-t.a);
    return{l:d<-.07,r:d>.07,f:Math.abs(d)<.7&&Math.hypot(p.x-t.x,p.y-t.y)>150,s:Math.abs(d)<.1};
  }
  const dc=Math.hypot(c.x-t.x,c.y-t.y);
  if(dc<=300)return roamStep(t);
  const tg=path(t,c)||c;
  const d=norm(Math.atan2(tg.y-t.y,tg.x-t.x)-t.a);
  return{l:d<-.07,r:d>.07,f:Math.abs(d)<.7};
}

// Hilfspanzer teilen sich (ohne Sammelbefehl) in Gruppen zu 3 bis 5 auf und ziehen einzeln umher
function regroup(now){
  if(now<rgNext)return;
  rgNext=now+1500;
  const L=tanks[me];
  if(!L||!L.alive)return;
  const al=tanks.filter((t,i)=>t.alive&&t.team===0&&t.ally&&!t.turret&&i!==me);
  const n=al.length,gc=Math.max(1,Math.ceil(n/5)),sz=Math.max(1,Math.ceil(n/gc));
  al.forEach((t,i)=>{t.gid=Math.floor(i/sz)});
  const [pr,pc]=cellOf(L),seen=flood(pr,pc);
  for(let g=0;g<gc;g++){
    const o=ggoal[g];
    if(o&&now<o.until)continue;
    for(let k=0;k<40;k++){
      const c=rnd(C),r=rnd(R),x=(c+.5)*S,y=(r+.5)*S,d=Math.hypot(x-L.x,y-L.y);
      if(seen[r*C+c]&&d>500&&d<1500){ggoal[g]={x,y,until:now+14000+rnd(6000)};break}
    }
  }
}
function roamStep(t){
  const g=ggoal[t.gid||0];
  if(!g)return{};
  if(Math.hypot(g.x-t.x,g.y-t.y)<130){g.until=0;return{}}
  const q=path(t,g)||g,d=norm(Math.atan2(q.y-t.y,q.x-t.x)-t.a);
  return{l:d<-.07,r:d>.07,f:Math.abs(d)<.7};
}
// Gegner-Salve auf Truppenansammlungen (nur wenn mindestens 3 deiner Einheiten dicht beieinander stehen)
function troopSalvo(t,i,now){

  if(t.ann){

      }
    }
    t.sv2=now+[0,20000,14000][diff];

  if(!p||Math.hypot(p.x-t.x,p.y-t.y)>700||!clear(t,p))return;

}

}

}

}

// Schlacht-Bot (beide Teams gleich stark): Abprall-Schüsse, kein Ausweichen
// Schwer: Gruppen kommen von verschiedenen Seiten auf den Schwerpunkt deines Teams zu (oben, Mitte, unten)
// Der Gegner prüft alle paar Sekunden die Lage und wechselt die Taktik: angreifen, zurückziehen oder von den Seiten umgehen
function reassess(now){
  if(now<etNext)return;
  etNext=now+[16000,11000,7000][diff]*(.8+Math.random()*.4);
  const en=tanks.filter(t=>t.alive&&t.team===1&&t.order),my=tanks.filter(t=>t.alive&&t.team===0);
  if(!en.length)return;
  const E=en.length,M=my.length;
  let tac=E>M*1.2?'attack':E<M*.7?'hold':(Math.random()<.5?'flank':'attack');
  if(diff===0&&tac==='flank')tac='attack';
  etac=tac;
  if(tac==='attack'){en.forEach(t=>{if(t.order==='hold'){t.order='attack';t.conv=true}})}
  else if(tac==='hold'){en.forEach(t=>{if(t.order==='attack'){const f=nearestRaw(t);if(!f||Math.hypot(f.x-t.x,f.y-t.y)>650){t.order='hold';t.flank=false}}})}
  else{en.forEach(t=>{if(t.order==='attack'&&Math.random()<.6){t.grp=rnd(3);t.flank=true;t.ft=now+7000}})}
  const el=document.getElementById('binfo');
  if(el)el.textContent='Gegner-Taktik: '+({attack:'greift an',hold:'zieht sich zurück und verteidigt',flank:'umgeht dich von den Seiten'})[tac];
}
function flankGoal(t){
  let sx=0,sy=0,n=0;
  for(const o of tanks)if(o.alive&&o.team===0){sx+=o.x;sy+=o.y;n++}
  if(!n)return null;
  const an=[-1.45,0,1.45][(t.grp||0)%3];
  return{x:Math.max(40,Math.min(AW-40,sx/n+Math.cos(an)*480)),y:Math.max(40,Math.min(AH-40,sy/n+Math.sin(an)*480))};
}

function battleAI(t){
  if(gather&&gq.includes(t)&&mode==='battle'&&phase==='fight')return followAI(t);
  const p=nearest(t);
  if(!p)return{};
  if(t.team===1&&(diff===2||t.melee)){const dg=dodge(t);if(dg)return dg}
  planAim(t,p);
  if(t.aim!=null){
    const d=norm(t.aim-t.a);
    return{l:d<-.03,r:d>.03,s:Math.abs(d)<.045};
  }
  if(t.flank){
    const gl=flankGoal(t);
    if(gl&&performance.now()<t.ft&&Math.hypot(gl.x-t.x,gl.y-t.y)>130){
      const q=path(t,gl)||gl,e=norm(Math.atan2(q.y-t.y,q.x-t.x)-t.a);
      return{l:e<-.07,r:e>.07,f:Math.abs(e)<.7};
    }
    t.flank=false;
  }
  if(t.order==='hold'){
    if(t.dest){
      const q=path(t,t.dest);
      if(q&&performance.now()<t.destT&&Math.hypot(t.dest.x-t.x,t.dest.y-t.y)>30){
        const e=norm(Math.atan2(q.y-t.y,q.x-t.x)-t.a);
        return{l:e<-.07,r:e>.07,f:Math.abs(e)<.7};
      }
      t.dest=null;
    }
    if(!clear(t,p))return{};
    const d=norm(Math.atan2(p.y-t.y,p.x-t.x)-t.a);
    return{l:d<-.05,r:d>.05,s:Math.abs(d)<.1};
  }
  return botOld(t,p);
}

// Mündung: liegt die Wand direkt vor dem Rohr, startet die Kugel vor der Wand statt dahinter (verhindert Schießen durch Wände)
function muzzle(t,a){
  let d=20;
  for(let s=4;s<=20;s+=2){if(wallAt(t.x+Math.cos(a)*s,t.y+Math.sin(a)*s)){d=Math.max(0,s-4);break}}
  return{x:t.x+Math.cos(a)*d,y:t.y+Math.sin(a)*d};
}

function wallAtOld(x,y){
  for(let i=0;i<rects.length;i++){
    const r=rects[i];
    if(x>r[0]-3&&x<r[0]+r[2]+3&&y>r[1]-3&&y<r[1]+r[3]+3)return true;
  }
  return false;
}

// Simuliert einen Schuss mit höchstens einem Abprall; true, wenn er das Ziel trifft und nicht den Schützen selbst
function shotHits(t,p,ang){
  let vx=Math.cos(ang)*4.2,vy=Math.sin(ang)*4.2,n=0;
  const b=muzzle(t,ang);
  for(let s=0;s<230;s++){
    let w=false;
    b.x+=vx;if(wallAt(b.x,b.y)){b.x-=vx;vx=-vx;w=true}
    b.y+=vy;if(wallAt(b.x,b.y)){b.y-=vy;vy=-vy;w=true}
    if(w&&++n>1)return false;
    if(n>0&&Math.hypot(t.x-b.x,t.y-b.y)<15)return false;
    if(Math.hypot(p.x-b.x,p.y-b.y)<15)return true;
  }
  return false;
}

function nearest(t){
  const b=nearestRaw(t);
  if(!b)return null;
  const k=t.tg;
  if(k&&k.alive&&k.team!==t.team&&k!==b){
    const dk=Math.hypot(k.x-t.x,k.y-t.y),db=Math.hypot(b.x-t.x,b.y-t.y);
    if(dk<db*1.35+60)return k;
  }
  t.tg=b;
  return b;
}
function nearestRaw(t){
  let best=null,bd=1e9;
  for(const o of tanks)if(o.alive&&o.team!==t.team){
    const d=Math.hypot(o.x-t.x,o.y-t.y);
    if(d<bd){bd=d;best=o}
  }
  return best;
}

// Gegner: Abprall-Schüsse mit Zielfehler und Denkpause
function botInput(t,p){
  const now=performance.now();
  if(!t.alive||!p.alive)return{};
  if(!(now<t.pn)){
    t.pn=now+(near(t)>=2?dd().pnn:dd().pn);t.aim=null;
    let bd=9;
    for(let i=0;i<90;i++){
      const ang=i/90*2*Math.PI;
      if(shotHits(t,p,ang)){const d=Math.abs(norm(ang-t.a));if(d<bd){bd=d;t.aim=ang+(Math.random()-.5)*dd().j}}
    }
  }
  if(t.aim!=null){
    const d=norm(t.aim-t.a);
    return{l:d<-.03,r:d>.03,s:Math.abs(d)<.045};
  }
  return botOld(t,p);
}

function botOld(t,p){
  const out={};
  if(!t.alive||!p.alive)return out;
  const los=clear(t,p);
  const tg=los?p:(path(t,p)||p);
  const d=norm(Math.atan2(tg.y-t.y,tg.x-t.x)-t.a);
  out.l=d<-.07;out.r=d>.07;
  out.f=Math.abs(d)<.7&&!(los&&Math.hypot(t.x-p.x,t.y-p.y)<90);
  out.s=los&&Math.abs(d)<.1;
  return out;
}

// Ausweichen: weicht seitlich aus, wenn eine feindliche Kugel auf ihn zukommt (nur etwa 8 % der Kugeln bemerkt er nicht)
function dodge(t){
  let best=null,bt=t.dh??(diff===2?90:70);
  for(const b of bullets){
    if(b.team===t.team||(mode==='battle'&&!clumpMode&&!t.melee&&b.ord!=='hold'))continue;
    const rx=t.x-b.x,ry=t.y-b.y,tc=(rx*b.vx+ry*b.vy)/(b.vx*b.vx+b.vy*b.vy);
    if(tc<=0||tc>=bt)continue;
    const cx=rx-b.vx*tc,cy=ry-b.vy*tc;
    if(Math.hypot(cx,cy)<(t.dr??(diff===2?40:32))){bt=tc;best={b,cx,cy}}
  }
  if(!best){t.thr=false;return null}
  if(!t.thr){t.thr=true;t.miss=Math.random()<(t.dm??(mode==='battle'?(clumpMode?.08:.25):dd().miss))}
  if(t.miss)return null;
  const {b,cx,cy}=best,L=Math.hypot(b.vx,b.vy);
  let px=-b.vy/L,py=b.vx/L;
  if(px*cx+py*cy<0){px=-px;py=-py}
  if(wallAt(t.x+px*34,t.y+py*34)){px=-px;py=-py;if(wallAt(t.x+px*34,t.y+py*34))return null}
  const d=norm(Math.atan2(py,px)-t.a);
  const rv=Math.abs(d)>1.7,e2=rv?norm(d+Math.PI):d;
  return{l:e2<-.1,r:e2>.1,f:!rv&&Math.abs(e2)<1.3,b:rv&&Math.abs(e2)<1.3};
}

// Hilfs-Bots: gehen auf verschiedene Seiten des Gegners, um ihn einzukreisen, und schießen bei freier Sicht
function ally(t){
  const e=nearest(t);
  if(!e)return{};
  const los=clear(t,e),dist=Math.hypot(e.x-t.x,e.y-t.y);
  const ref=tanks[me].alive?tanks[me]:t;
  const a0=Math.atan2(ref.y-e.y,ref.x-e.x)+(t.side?1:-1)*2.1;
  const goal={x:Math.min(AW-30,Math.max(30,e.x+Math.cos(a0)*170)),y:Math.min(AH-30,Math.max(30,e.y+Math.sin(a0)*170))};
  const tg=los?e:(nextCell(t,goal)||nextCell(t,e)||e);
  const d=norm(Math.atan2(tg.y-t.y,tg.x-t.x)-t.a);
  return{l:d<-.07,r:d>.07,f:Math.abs(d)<.7&&!(los&&dist<220),s:los&&Math.abs(d)<.1};
}

function input(i){
  const t=tanks[i];
  if(i===me){
    const c=ctl[0];
    if(joy.s){const d=norm(joy.ang-t.a),w=Math.max(-1,Math.min(1,d*3))*1.8;return{f:joy.s*Math.max(0,Math.cos(d)),l:Math.max(-w,0),r:Math.max(w,0),s:keys[c.s]}}
    return{f:keys[c.f],b:keys[c.b],l:keys[c.l],r:keys[c.r],s:keys[c.s]};
  }
  if(mode==='base')return t.team===1?baseAI(t):baseAlly(t);
  if(mode==='endless')return t.team===1?endAI(t):allyE(t);
  if(mode==='battle')return battleAI(t);
  if(t.team===1){
    if(!bot){const c=ctl[1];return{f:keys[c.f],b:keys[c.b],l:keys[c.l],r:keys[c.r],s:keys[c.s]}}
    const p=nearest(t);
    return p?(dodge(t)||botInput(t,p)):{};
  }
  return ally(t);
}

function update(k,now){
  if((mode==='battle'||mode==='base')&&phase==='place')return;
  planCnt=0;
  if(mode==='battle'&&phase==='fight'&&!resetAt)releaseHolds(now);
  if(!resetAt){
    tanks.forEach((t,i)=>{
      if(!t.alive)return;
      const c=input(i),boost=(mode==='endless'||mode==='base')?(gather&&gq.includes(t)&&i!==me?(t.sp||1.5):(t.spd||1)):t.team===1&&bot&&mode==='duel'?dd().sp:(gather&&mode==='battle'&&gq.includes(t)&&i!==me?(t.sp||1.5):(mode==='battle'&&t.team===1?dd().bsp*(t.tspd||1):(mode==='duel'&&t.ally&&diff===2?.9:1)));
      t.a+=((+c.r||0)-(+c.l||0))*.055*k*boost;
      const m=((+c.f||0)-(+c.b||0))*2.4*k*boost;
      t.x+=Math.cos(t.a)*m;t.y+=Math.sin(t.a)*m;
      push(t,13);
      if(mode==='duel'&&bot&&t.team===1&&diff>0)trySalvo(t,i,now);
      if(mode==='battle'&&t.team===1&&diff===2)salvoB(t,i,now);
      if(mode==='endless'&&t.elite)trySalvo(t,i,now);
      if(mode==='endless'&&t.team===1&&t.salv)troopSalvo(t,i,now);
      if(c.s&&now>t.cd&&bullets.filter(b=>b.o===i).length<5&&(i===me||t.mount||(t.team===1&&mode==='duel')||safeShot(t))){
        const human=i===me||(mode==='duel'&&t.team===1&&!bot);
        t.cd=now+(human?300:(mode==='endless'||mode==='base')?(t.cdn||500):mode==='battle'?(t.team===1?dd().bcd*(t.tcd||1):800):t.team===1?(near(t)>=2?dd().cdn:dd().cd):(diff===2?850:600));
        if(i===me)sfx('shot');const mz=muzzle(t,t.a);bullets.push({x:mz.x,y:mz.y,vx:Math.cos(t.a)*4.2,vy:Math.sin(t.a)*4.2,o:i,team:t.team,ord:t.order,mine:i===me,mb:t.mb,dmg:t.dmg,gh:!!t.mount,n:0,die:now+7000});
      }
    });
  }
  if(!resetAt)for(let i=0;i<tanks.length;i++)for(let j=i+1;j<tanks.length;j++){
    const a=tanks[i],c=tanks[j];
    if(!a.alive||!c.alive||a.team!==c.team||a.turret||c.turret)continue;
    const dx=c.x-a.x,dy=c.y-a.y,d=Math.hypot(dx,dy);
    if(d<26&&d>.01){
      const s=Math.min(1.2*k,(26-d)/2);
      a.x-=dx/d*s;a.y-=dy/d*s;c.x+=dx/d*s;c.y+=dy/d*s;
      push(a,13);push(c,13);
    }
  }
  if(mode==='endless'&&!resetAt)endlessTick(now);
  if(mode==='base'&&!resetAt){baseTick(now);checkGrow()}
  if(mode==='battle'&&phase==='fight'&&!resetAt)reassess(now);
  bullets=bullets.filter(b=>now<b.die);
  for(const b of bullets){
    let bounced=false;
    b.x+=b.vx*k;if(inWall(b)){bwHit(b);b.x-=b.vx*k;b.vx*=-1;bounced=true}
    b.y+=b.vy*k;if(inWall(b)){bwHit(b);b.y-=b.vy*k;b.vy*=-1;bounced=true}
    if(bounced&&++b.n>(b.mb||1))b.die=0;

      if(t.alive&&!t.mount&&(t.team!==b.team||(ti===b.o&&ti!==me))&&Math.hypot(t.x-b.x,t.y-b.y)<(t.hr||15)){if(absorb(t,ti,b))return false;if(t.hp>1){t.hp--;t.hit=performance.now()+180;b.die=0;sfx('kill');return true}t.alive=false;b.die=0;if(b.mine&&t.team!==b.team)myKills++;sfx(ti===me?'hit':'kill');if(ti===me)vib(200);return true}
    });
  }
  if(!resetAt){
    if(mode!=='endless'&&mode!=='base'&&!tanks[me].alive){const nx=tanks.findIndex(t=>t.alive&&t.team===0);if(nx>=0)me=nx}
    const mine=tanks.some(t=>t.alive&&t.team===0),foe=tanks.some(t=>t.alive&&t.team===1);
    if(mode==='base'?!(tanks.find(t=>t.tower)||{alive:true}).alive:mode==='endless'?!tanks[me].alive:(!mine||!foe)){
      if(mine)score[0]++;else if(foe)score[1]++;
      msg=(mode==='endless'||mode==='base')?'Game Over bei Welle '+wave+' (Kills: '+kills+')':!mine&&!foe?'Unentschieden':mode==='battle'?(mine?'Dein Team gewinnt!':'Das Gegner-Team gewinnt'):mine?'Punkt für Spieler 1':'Punkt für '+(bot?'den Bot':'Spieler 2');
      resetAt=now+1800;updateScore();recEnd(mine);sfx(mode==='endless'||mode==='base'||!mine?'over':'win');vib(mode==='endless'||mode==='base'||!mine?300:100);
    }
  }
  if(resetAt&&now>resetAt){newRound();if(mode==='endless'||mode==='base'){running=false;paused=true;showStart(mode)}}
}
